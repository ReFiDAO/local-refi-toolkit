│   ├── tools/            # Software, configs, scripts
│   ├── assets/           # Graphics, presentations
│   └── examples/         # Sample data, filled forms
└── support/              # Support materials
    ├── checklists/       # Verification lists
    ├── worksheets/       # Planning sheets
    ├── guides/           # How-to sub-guides
